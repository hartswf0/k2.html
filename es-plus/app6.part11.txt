function bankUid(){return `bank_${Date.now().toString(36)}_${Math.floor(Math.random()*1e6).toString(36)}`}
function compactSound(s){
  if(!s)return null;
  return{name:s.name||'',path:s.path||'',tempo:s.tempo??null,key:s.key??null,instrument:s.instrument??'',genre:s.genre??'',artist:s.artist??''};
}
function compactEventForBank(e){
  const x={id:e.id,type:e.type,track:e.track,start:+e.start,dur:+e.dur,note:e.note??0,vel:+e.vel,transpose:+(e.transpose||0),role:e.role||'',query:e.query||'',behavior:e.behavior||'',muted:!!e.muted};
  if(e.sound)x.sound=compactSound(e.sound);
  if(e.degree!=null)x.degree=e.degree;
  if(e.family)x.family=e.family;
  return x;
}
function harvestSelectedParts(){
  if(!state.ir)return setTrace('HARVEST','RUN first','no audible arrangement');
  if(state.partStonePreview>=0)return setTrace('HARVEST','preview not committed','USE the part stone first');
  const tracks=selectedArray();
  if(!tracks.length)return setTrace('HARVEST','select track numbers','nothing selected');
  const s=current(),rows=partBankRows(),made=[];
  for(const tr of tracks){
    const events=state.ir.events.filter(e=>e.track===tr);
    if(!events.length)continue;
    const rec={
      id:bankUid(),savedAt:new Date().toISOString(),
      parentStoneId:s?.id||null,parentSong:s?.name||'untitled',parentGenre:s?.genre||state.plan?.genre||'hybrid',
      parentPrompt:s?.source||E.prompt.value||'',promptLogIds:[...(s?.promptLogIds||[])],
      sourceTrack:tr,trackName:trackName(tr),tempo:state.ir.tempo||state.plan?.tempo||100,
      key:state.plan?.key||s?.key||'C',mode:state.plan?.mode||s?.mode||'minor',bars:state.ir.bars||state.plan?.bars||8,
      family:state.ir.family||family(state.plan?.genre||'',state.plan?.palette||[]),
      patch:cloneValue(trackPatch(tr)),
      events:events.map(compactEventForBank)
    };
    rows.push(rec);made.push(rec);
  }
  writePartBank(rows);
  for(const r of made)state.partBankSelected.add(r.id);
  renderPartBank();
  setTrace('HARVEST',made.map(x=>x.trackName).join(' + '),`${made.length} part${made.length===1?'':'s'} saved from ${s?.name||'song'}`);
  if(E.trackReadout)E.trackReadout.textContent=`HARVESTED ${made.length} · PARTS bank now has ${partBankRows().length}`;
}

function transformedFusionEvents(parent,tf,newTrack,targetTempo,targetBars,index){
  const out=[],sourceTempo=Math.max(1,+parent.tempo||targetTempo),tempoScale=targetTempo/sourceTempo,maxBeat=targetBars*4;
  for(const e0 of parent.events){
    const density=clamp(+tf.density||1,.15,1),keep=(hash(`${parent.id}|${e0.id}|${index}|fusion-density`)%10000)/10000;if(keep>density)continue;
    const start=((+e0.start||0)*tempoScale+(+tf.shift||0));if(start<0||start>=maxBeat)continue;
    const e={...cloneValue(e0),id:`f${index}_${e0.id}_${out.length}`,track:newTrack,start,dur:Math.max(.02,(+e0.dur||.1)*tempoScale*clamp(+tf.durationScale||1,.35,2.5)),vel:clamp((+e0.vel||.5)*clamp(+tf.gain||1,.1,1.5),.01,1),muted:false};
    const trans=+tf.transpose||0;
    if(Number.isFinite(+e.note)&&e.type!=='kick'&&e.type!=='snare'&&e.type!=='hat')e.note=+e.note+trans;
    if(e.type==='sample')e.transpose=(+e.transpose||0)+trans;
    out.push(e);
  }
  return out;
}
function compileFusionStone(stone,parents){
  const byId=new Map(parents.map(x=>[x.id,x])),events=[],names={},used=[];
  let nextTrack=21,index=0;
  for(const tf of stone.transforms){
    const parent=byId.get(tf.partId);if(!parent)continue;
    const tr=nextTrack++;names[tr]=`${parent.trackName} ← ${parent.parentSong}`;used.push(parent.id);
    events.push(...transformedFusionEvents(parent,tf,tr,stone.tempo,stone.bars,index++));
  }
  if(used.length<2)throw Error('fusion stone did not preserve at least two parents');
  events.sort((a,b)=>a.start-b.start||a.track-b.track);
  const plan={genre:'fusion',tempo:stone.tempo,key:stone.key,mode:stone.mode,bars:stone.bars,swing:0,groove:'free_pulse',density:1,palette:[],progression:[1],bass:[0],lead:[0],surprise:stone.concept,trackPatches:{},fusionMode:true,fusionTransforms:cloneValue(stone.transforms)};
  return{plan,ir:{...plan,family:'fusion',events,length:stone.bars*4},names};
}
function fusionCode(stone){
  const lines=[`# ES+ FUSION ${stone.name}`,`# ${stone.concept}`,`fusionMode(true)`,`tempo(${stone.tempo})`,`key(${JSON.stringify(stone.key)}, ${JSON.stringify(stone.mode)})`,`bars(${stone.bars})`];
  for(const tf of stone.transforms)lines.push(`fusionPart(${JSON.stringify(tf.partId)}, ${JSON.stringify({transpose:tf.transpose,shift:tf.shift,gain:tf.gain,density:tf.density,durationScale:tf.durationScale})})`);
  return lines.join('\n');
}

const parseCodeFusionBase=parseCode;
parseCode=function(src){
  const parts=[],clean=[];let fusionMode=false;
  for(const line0 of src.split('\n')){
    const line=line0.replace(/#.*$/,'').trim();
    if(!line){clean.push(line0);continue}
    let m=line.match(/^fusionMode\((.*)\)$/);if(m){fusionMode=String(m[1]).trim()!=='false';continue}
    m=line.match(/^fusionPart\((.*)\)$/);if(m){const a=splitArgs(m[1]).map(parseVal);parts.push({partId:String(a[0]),...(typeof a[1]==='object'?a[1]:{})});continue}
    clean.push(line0);
  }
  const p=parseCodeFusionBase(clean.join('\n'));p.fusionMode=fusionMode;p.fusionParts=parts;return p;
};
