function regionPatchSchema(){return{type:'object',additionalProperties:false,required:['track','gainStart','gainEnd','densityStart','densityEnd','durationScale','transpose','shift','query'],properties:{track:{type:'integer',minimum:1,maximum:96},gainStart:{type:'number',minimum:0,maximum:1.6},gainEnd:{type:'number',minimum:0,maximum:1.6},densityStart:{type:'number',minimum:.02,maximum:1},densityEnd:{type:'number',minimum:.02,maximum:1},durationScale:{type:'number',minimum:.15,maximum:4},transpose:{type:'integer',minimum:-24,maximum:24},shift:{type:'number',minimum:-4,maximum:4},query:{type:'string',maxLength:70}}}}
function regionStoneSchema(){return{type:'object',additionalProperties:false,required:['stones'],properties:{stones:{type:'array',minItems:4,maxItems:4,items:{type:'object',additionalProperties:false,required:['name','concept','patches'],properties:{name:{type:'string',maxLength:28},concept:{type:'string',maxLength:120},patches:{type:'array',minItems:1,maxItems:36,items:regionPatchSchema()}}}}}}}
function regionBaseSnapshot(){return{code:E.code.value,plan:cloneValue(state.plan),ir:cloneValue(state.ir),selected:new Set(state.selectedTracks),selectedTrack:state.selectedTrack,soloActive:state.soloActive}}
async function compileRegionStone(stone){if(stone.previewIR&&stone.previewPlan&&stone.previewCode)return stone;const code=regionCandidateCode(state.regionStoneBase.code,stone),plan=parseCode(code),ir=await arrange(plan);stone.previewCode=code;stone.previewPlan=cloneValue(plan);stone.previewIR=cloneValue(ir);return stone}
function showRegionPreview(stone,index){state.regionStonePreview=index;state.plan=cloneValue(stone.previewPlan);state.ir=cloneValue(stone.previewIR);state.selectedTracks=new Set(state.regionStoneTargets);state.selectedTrack=state.regionStoneTargets[0]??null;renderTimeline();drawSpec();updateSelectionUI();renderRegionStones();refreshRegionUI();E.codeState.textContent='region preview · code unchanged';E.trackReadout.textContent=`PREVIEW ${index+1} · ${stone.name} — ${stone.concept}`;setTrace('REGION STONE',`${index+1}/4 ${stone.name}`,'preview only · USE to commit')}
async function previewRegionStone(index,autoPlay=false){const s=state.regionStones[index];if(!s||!state.regionStoneBase)return;if(state.playing)stop();try{await compileRegionStone(s);showRegionPreview(s,index);if(autoPlay)await auditionRegion()}catch(e){E.trackReadout.textContent=e.message;setTrace('FAULT','region stone preserved',e.message)}}
function restoreRegionOriginal(autoPlay=false){const b=state.regionStoneBase;if(!b)return;if(state.playing)stop();E.code.value=b.code;state.plan=cloneValue(b.plan);state.ir=cloneValue(b.ir);state.selectedTracks=new Set(b.selected);state.selectedTrack=b.selectedTrack;state.soloActive=b.soloActive;state.regionStonePreview=-1;renderTimeline();drawSpec();updateSelectionUI();renderRegionStones();refreshRegionUI();updateGutter();E.codeState.textContent='matches sound';E.trackReadout.textContent='ORIGINAL REGION · no candidate committed';if(autoPlay)auditionRegion()}

async function generateRegionStones(request=''){
  const r=state.regionSelection;if(!r)return;const targets=regionTargets();if(!targets.length){E.trackReadout.textContent='no editable target tracks';return}if(!state.connected){showSheet();return}if(state.playing)stop();
  if(state.partStones?.length)clearPartStones({restore:false,hide:true});clearRegionStones(false);state.regionStoneBase=regionBaseSnapshot();state.regionStoneTargets=[...targets];
  const prompt=request.trim()||`Give me four strong alternatives for this ${r.label.toLowerCase()} without changing anything outside the circled region.`;const anchors=[...state.anchorTracks].filter(t=>allTracks().includes(t)),inside=targets.map(tr=>({track:tr,name:trackName(tr),events:(state.ir.events||[]).filter(e=>e.track===tr&&e.start>=r.start&&e.start<r.end).slice(0,36).map(e=>({type:e.type,start:+e.start.toFixed(2),dur:+e.dur.toFixed(2),note:e.note,vel:+e.vel.toFixed(2),role:e.role,query:e.query,sound:e.sound?.name}))}));
  E.trackAI.disabled=true;E.trackAI.textContent='CASTING';E.trackReadout.textContent='casting 4 region stones…';
  try{
    const d=await apiFetch('/responses',{method:'POST',body:JSON.stringify({model:state.model,instructions:'You are editing a circled TIME REGION of a multitrack musical arrangement. Return exactly four audibly distinct REGION STONES. You may modify TARGET tracks only, and only inside the supplied [start,end) region. INVARIANT tracks and all music outside the region are immutable evidence. Shape rises/drops/breaks through gain and event-density ramps, timing, note duration, transposition, and sample-material query. Preserve continuity at the boundaries unless the request explicitly wants rupture. Do not make four cosmetic nudges; make four different formal hypotheses.',input:[{role:'user',content:[{type:'input_text',text:`SONG ${state.plan?.genre||''} | ${state.plan?.tempo||''} bpm | ${state.plan?.key||''} ${state.plan?.mode||''}\nREGION ${JSON.stringify({start:r.start,end:r.end,label:r.label,energy:r.energy})}\nREQUEST ${prompt}\nTARGETS ${JSON.stringify(inside)}\nINVARIANTS ${JSON.stringify(anchors.map(trackContext))}\nOUTSIDE REGION: immutable. Return exactly FOUR region stones.`}]}],text:{format:{type:'json_schema',name:'region_stones',strict:true,schema:regionStoneSchema()}}})});
    const parsed=JSON.parse(outputText(d)),targetSet=new Set(targets);state.regionStones=(parsed.stones||[]).slice(0,4).map((s,i)=>({id:`region_${Date.now().toString(36)}_${i}`,name:s.name,concept:s.concept,patches:(s.patches||[]).filter(x=>targetSet.has(Number(x.track))&&!state.anchorTracks.has(Number(x.track)))})).filter(s=>s.patches.length);if(state.regionStones.length!==4)throw Error(`received ${state.regionStones.length}/4 usable region stones`);
    const row=recordPair({scope:'region-stones',prompt,targets,anchors,region:cloneValue(r),stoneId:current()?.id,completion:{summary:`4 region stones for ${regionText(r)}`,regionStones:state.regionStones.map(s=>({id:s.id,name:s.name,concept:s.concept,patches:s.patches}))}});state.regionStoneLogId=row.id;renderRegionStones();E.trackReadout.textContent='4 REGION STONES · tap to audition · USE commits only that region';setTrace('4 REGION STONES',regionText(r),'tap to audition')
  }catch(e){restoreRegionOriginal(false);clearRegionStones(false);E.trackReadout.textContent=e.message;setTrace('FAULT','region prompt preserved',e.message)}finally{E.trackAI.disabled=false;E.trackAI.textContent='CAST 4'}
}
