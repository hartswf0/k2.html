function snapBeat(x){return Math.round(x*4)/4}
function regionFromGesture(path){
  if(!state.ir||!path?.length)return null;const r=E.spec.getBoundingClientRect(),xs=path.map(p=>p.x),min=Math.min(...xs),max=Math.max(...xs);if(max-min<10)return null;
  let start=snapBeat(clamp(min/r.width*state.ir.length,0,state.ir.length)),end=snapBeat(clamp(max/r.width*state.ir.length,0,state.ir.length));
  if(end-start<.5)end=Math.min(state.ir.length,start+.5);return{start:+start.toFixed(2),end:+end.toFixed(2),label:'REGION',energy:[]};
}
function regionEnergy(selection=state.regionSelection,ir=state.ir){
  if(!selection||!ir)return{label:'REGION',bins:[0,0,0,0]};const bins=[0,0,0,0],span=Math.max(.001,selection.end-selection.start);
  for(const e of ir.events){if(e.muted||e.start<selection.start||e.start>=selection.end)continue;const i=clamp(Math.floor((e.start-selection.start)/span*4),0,3);bins[i]+=(e.vel||.5)*Math.min(1,Math.max(.2,e.dur||.1))}
  const mx=Math.max(.001,...bins),n=bins.map(x=>x/mx),first=(n[0]+n[1])*.5,last=(n[2]+n[3])*.5,mid=(n[1]+n[2])*.5,edges=(n[0]+n[3])*.5;let label='HOLD';
  if(last>first+.28)label='RISE';else if(first>last+.28)label='DROP';else if(mid<edges-.3)label='BREAK';else if(Math.max(...n)-Math.min(...n)>.35)label='ENERGY';
  return{label,bins:n.map(x=>+x.toFixed(2))};
}
function regionText(r=state.regionSelection){if(!r)return'';return `${r.label} ${r.start.toFixed(2)}–${r.end.toFixed(2)}b`}

  }

}

function drawRegionOverlay(){
  if(!state.ir)return;const c=E.spec.getContext('2d'),r=E.spec.getBoundingClientRect();if(r.width<1||r.height<1)return;const orange=getComputedStyle(document.documentElement).getPropertyValue('--orange').trim()||'#d85e25';
  if(state.regionSelection){const x=state.regionSelection.start/state.ir.length*r.width,w=(state.regionSelection.end-state.regionSelection.start)/state.ir.length*r.width;c.save();c.fillStyle='rgba(216,94,37,.12)';c.fillRect(x,0,w,r.height);c.strokeStyle=orange;c.lineWidth=2;c.strokeRect(x+1,1,Math.max(1,w-2),Math.max(1,r.height-2));c.fillStyle=orange;c.font='700 9px ui-monospace,monospace';c.fillText(regionText(),x+5,16);c.restore()}
  if(state.regionGesture.length>1){c.save();c.strokeStyle=orange;c.lineWidth=2;c.beginPath();c.moveTo(state.regionGesture[0].x,state.regionGesture[0].y);for(const p of state.regionGesture.slice(1))c.lineTo(p.x,p.y);c.stroke();c.restore()}
}
