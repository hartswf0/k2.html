function parseCode(src){const p={genre:"hybrid",tempo:100,key:"C",mode:"minor",swing:0,bars:8,groove:"backbeat",density:.65,palette:[],progression:[1,4,6,5],bass:[0,0,3,5],lead:[0,2,4,7,5],surprise:"",drawRhythm:null,drawContour:null,drawEnergy:null,drawBlank:null,trackPatches:{}};for(const line0 of src.split("\n")){const line=line0.replace(/#.*$/,"").trim();if(!line)continue;const m=line.match(/^([A-Za-z_]\w*)\((.*)\)$/);if(!m)throw Error(`cannot read: ${line.slice(0,28)}`);const a=splitArgs(m[2]).map(parseVal);switch(m[1]){case"genre":p.genre=String(a[0]);break;case"tempo":p.tempo=clamp(+a[0],40,220);break;case"key":p.key=String(a[0]);p.mode=String(a[1]);break;case"groove":p.groove=String(a[0]);p.swing=clamp(+a[1]||0,0,.45);break;case"bars":p.bars=clamp(Math.round(+a[0]),2,32);break;case"density":p.density=clamp(+a[0],.05,1);break;case"palette":p.palette.push({role:String(a[0]),query:String(a[1]),behavior:String(a[2]),level:clamp(+a[3],.05,1)});break;case"progression":p.progression=Array.isArray(a[0])?a[0]:p.progression;break;case"bass":p.bass=Array.isArray(a[0])?a[0]:p.bass;break;case"lead":p.lead=Array.isArray(a[0])?a[0]:p.lead;break;case"drawRhythm":p.drawRhythm=String(a[0]).split("").map(x=>x==="1"?1:0);break;case"drawContour":p.drawContour=Array.isArray(a[0])?a[0]:null;break;case"drawEnergy":p.drawEnergy=Array.isArray(a[0])?a[0]:null;break;case"drawBlank":p.drawBlank=Array.isArray(a[0])?a[0]:null;break;case"surprise":p.surprise=String(a[0]||"");break;case"track":p.trackPatches[String(Math.round(+a[0]))]={...(p.trackPatches[String(Math.round(+a[0]))]||{}),...(typeof a[1]==="object"?a[1]:{})};break;default:throw Error(`unknown ${m[1]}()`)} }return p}
function setTrackPatch(track,patch){const src=E.code.value.split("\n"),re=new RegExp(`^\\s*track\\(\\s*${track}\\s*,`),idx=src.findIndex(x=>re.test(x));let old={};if(idx>=0){try{old=parseVal(splitArgs(src[idx].match(/^\s*track\((.*)\)\s*$/)[1])[1])||{}}catch(_){old={}}}const merged={...old,...patch};if(merged.delete===false)delete merged.delete;if(merged.mute===false)delete merged.mute;const line=`track(${track}, ${JSON.stringify(merged)})`;if(idx>=0)src[idx]=line;else src.push(line);E.code.value=src.join("\n");E.codeState.textContent="changed";state.dirty=true;updateGutter();run()}

function family(genre,palette=[]){const s=(genre+" "+palette.map(p=>p.query).join(" ")).toLowerCase();if(/dub|reggae/.test(s))return"dub";if(/gospel|soul|funk|r&b|motown/.test(s))return"soul";if(/techno|house|garage|electro|club/.test(s))return"electronic";if(/ambient|drone|electroacoustic|field/.test(s))return"ambient";if(/gamelan|bali|java|gong/.test(s))return"gamelan";if(/afro|mbira|kora|gnawa|morocc|oud|world|latin|samba/.test(s))return"world";if(/rock|psych|guitar|punk|surf/.test(s))return"rock";if(/jazz|bebop|swing/.test(s))return"jazz";if(/hip.?hop|trap|boom.?bap/.test(s))return"hiphop";return"hybrid"}
function degreeMidi(p,deg,oct=4){const mode=MODES[p.mode]||MODES.minor,root=ROOTS[p.key]??0,d=Math.max(1,Math.round(deg))-1,scale=mode[d%mode.length]+12*Math.floor(d/mode.length);return 12*(oct+1)+root+scale}
function drawingGate(p,step){return!p.drawRhythm?.length?1:(p.drawRhythm[step%p.drawRhythm.length]?1:.28)}
function sectionEnergy(p,bar){if(p.drawEnergy?.length)return clamp(+p.drawEnergy[Math.floor(bar/p.bars*p.drawEnergy.length)]||.5,.15,1);return .55+.3*Math.sin((bar+1)/p.bars*Math.PI)}
function blankSection(p,bar){return p.drawBlank?.length?!!p.drawBlank[Math.floor(bar/p.bars*p.drawBlank.length)]:false}
function drumPattern(g){const K={backbeat:[[0,8],[4,12],[2,6,10,14]],shuffle:[[0,7,10],[4,12],[2,5,8,11,14]],broken:[[0,6,11],[4,13],[2,7,10,15]],halftime:[[0,10],[8],[2,6,10,14]],motorik:[[0,4,8,12],[4,12],[2,6,10,14]],clave:[[0,6,10],[4,12],[0,3,6,10,12]],garage:[[0,7,10],[4,12],[2,6,10,14]],triplet:[[0,8],[5,13],[0,3,6,9,12,15]],afro_lilt:[[0,6,10,14],[4,12],[1,4,7,10,13]],dub_space:[[0,10],[12],[3,7,11,15]],free_pulse:[[0,9],[5,13],[1,6,12]]};return K[g]||K.backbeat}
async function resolvePlanPalette(p){await loadCatalog();p.palette=p.palette.map((x,i)=>{const patch=p.trackPatches[String(7+i)]||{},query=patch.query||x.query,role=patch.role||x.role,soundName=patch.soundName;let sound=soundName?(state.catalog||[]).find(s=>s.name===soundName):null;if(!sound)sound=resolveSound(role,query,p.genre);if(sound)state.usedSounds.add(sound.name);return{...x,query,role,sound}})}
function applyPatches(events,p){const out=[];for(const e0 of events){const patch=p.trackPatches[String(e0.track)]||{};if(patch.delete||patch.mute)continue;const e={...e0,start:Math.max(0,e0.start+(+patch.shift||0)),vel:clamp(e0.vel*(patch.gain==null?1:+patch.gain),0,1)};if(Number.isFinite(+patch.transpose)&&e.note)e.note+=+patch.transpose;if(e.type==="sample"){e.transpose=+patch.transpose||0;if(patch.query)e.query=patch.query;if(patch.soundName){const s=(state.catalog||[]).find(x=>x.name===patch.soundName);if(s)e.sound=s}}out.push(e)}for(const [trk,patch] of Object.entries(p.trackPatches)){if(!patch.copyTo)continue;const src=Number(trk),dst=Number(patch.copyTo),shift=+patch.copyShift||0,trans=+patch.copyTranspose||0;for(const e0 of out.filter(e=>e.track===src)){const e={...e0,id:`${e0.id}c${dst}`,track:dst,start:Math.max(0,e0.start+shift)};if(e.note)e.note+=trans;if(e.type==="sample")e.transpose=(e.transpose||0)+trans;out.push(e)}}out.sort((a,b)=>a.start-b.start||a.track-b.track);return out}
async function arrange(p){await resolvePlanPalette(p);const events=[],fam=family(p.genre,p.palette),seed=hash(JSON.stringify({...p,trackPatches:undefined})),r=seeded(seed),dr=drumPattern(p.groove),add=(type,track,start,dur,note=0,vel=.7,meta={})=>events.push({id:`e${events.length+1}`,type,track,start,dur,note,vel,...meta});for(let bar=0;bar<p.bars;bar++){const base=bar*4,en=sectionEnergy(p,bar),blank=blankSection(p,bar);if(blank&&bar%2===1)continue;dr[0].forEach(st=>{if(r()<p.density*en*drawingGate(p,st))add("kick",1,base+st/4,.12,36,.8*en)});dr[1].forEach(st=>{if(r()<Math.min(1,.7+p.density*.3))add("snare",2,base+st/4,.1,38,.68*en)});dr[2].forEach(st=>{if(r()<p.density*en*drawingGate(p,st))add("hat",3,base+st/4,.05,42,.3+.3*en)});const deg=p.progression[bar%p.progression.length]||1;add("chord",5,base,.9,degreeMidi(p,deg,4),.36*en,{degree:deg,family:fam});const bm=p.bass.length?p.bass:[0];for(let i=0;i<4;i++){const st=i*4;if(r()<(.55+p.density*.35)*drawingGate(p,st))add("bass",4,base+i,.55,degreeMidi(p,deg,2)+(bm[(bar*2+i)%bm.length]||0),.55*en,{family:fam})}const lm=p.lead.length?p.lead:[0,2,4,5],cont=p.drawContour?.length?p.drawContour:null;for(let i=0;i<4;i++){const st=i*4;if(r()<p.density*.52*en*drawingGate(p,st))add("lead",6,base+i+.25,.35,degreeMidi(p,1,5)+(lm[(bar*3+i)%lm.length]||0)+(cont?cont[(bar+i)%cont.length]:0),.38*en,{family:fam})}}p.palette.forEach((x,idx)=>{const patch=p.trackPatches[String(7+idx)]||{},behavior=patch.behavior||x.behavior,every=behavior==="loop"?4:behavior==="punctuate"?3:behavior==="chop"?2:Math.max(4,p.bars);for(let bar=idx%2;bar<p.bars;bar+=every)add("sample",7+idx,bar*4,behavior==="one_shot"?.8:3.8,0,x.level,{paletteIndex:idx,role:x.role,query:x.query,behavior,sound:x.sound})});if(p.drawRhythm)for(let st=0;st<16;st++)if(p.drawRhythm[st])add("draw",20,st/4,.04,72+(p.drawContour?.[st%8]||0),.22);return{...p,family:fam,events:applyPatches(events,p),length:p.bars*4}}
async function run(){stop();try{state.plan=parseCode(E.code.value);state.ir=await arrange(state.plan);state.dirty=false;renderTimeline();drawSpec();E.codeState.textContent="matches sound";E.soundState.textContent=`${state.ir.events.length} events · ${state.ir.family}`;E.playBtn.disabled=!state.ir.events.length;E.mobileSound.disabled=false;E.exportBtn.disabled=false;if(mobile())showMobile("sound")}catch(e){E.codeState.textContent="fix";setTrace("FAULT","code preserved",e.message)}}
E.runBtn.onclick=run;

function trackName(tr){if(tr===1)return"KICK";if(tr===2)return"SNARE";if(tr===3)return"HAT";if(tr===4)return"BASS";if(tr===5)return"CHORDS";if(tr===6)return"LEAD";if(tr===20)return"DRAW";const p=state.plan?.palette?.[tr-7];return p?.role||`TRACK ${tr}`}
